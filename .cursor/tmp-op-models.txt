model Invoice {
  id                        String              @id @default(cuid())
  invoiceNumber             String
  title                     String?
  orderNumber               String?
  clientId                  String
  createdById               String
  issueDate                 DateTime
  dueDate                   DateTime
  subtotal                  Decimal             @db.Decimal(18, 2)
  taxAmount                 Decimal             @db.Decimal(18, 2)
  total                     Decimal             @db.Decimal(18, 2)
  status                    String              @default("draft")
  notes                     String?
  createdAt                 DateTime            @default(now())
  updatedAt                 DateTime            @updatedAt
  tenantId                  String
  discount                  Decimal             @default(0) @db.Decimal(18, 2)
  lastPaymentDate           DateTime?
  originalTotal             Decimal             @default(0) @db.Decimal(18, 2)
  refundAmount              Decimal             @default(0) @db.Decimal(18, 2)
  refundReason              String?
  refundedAt                DateTime?
  refundedById              String?
  remainingBalance          Decimal             @default(0) @db.Decimal(18, 2)
  totalDiscountAmount       Decimal             @default(0) @db.Decimal(18, 2)
  totalPaid                 Decimal             @default(0) @db.Decimal(18, 2)
  voidReason                String?
  voidedAt                  DateTime?
  voidedById                String?
  deletionReason            String?
  deletedAt                 DateTime?
  deletedById               String?
  isDeleted                 Boolean             @default(false)
  branchId                  String?
  footerPhoneOverride       String? // Override default phone in document footer
  footerBankDetailsOverride String? // Override default bank details in document footer
  // Reversal fields
  isReversal                Boolean             @default(false)
  reversedTransactionId     String?
  reversalReason            String?
  reversedAt                DateTime?
  reversedById              String?
  /// True when this invoice was created from Rental & Hiring (operational layer).
  isRentalInvoice           Boolean             @default(false)
  rentalTransaction         RentalTransaction?

  @@unique([tenantId, invoiceNumber])
  @@index([branchId])
  @@index([isDeleted])
  @@index([deletedAt])
  @@index([isReversal])
  @@index([reversedTransactionId])
  @@index([isRentalInvoice])
}

model PaymentAccount {
  id                      String              @id @default(cuid())
  tenantId                String
  name                    String
  accountType             String // Cash, Bank, Mobile Money, Wallet, POS Terminal
  reference               String? // Account number, wallet ID, etc.
  isActive                Boolean             @default(true)
  isSystem                Boolean             @default(false) // System accounts (like Cash) cannot be deleted
  /// Links this POS/payment method to a Chart of Accounts asset (cash/bank) so GL and A/L stay in sync
  coaAccountId            String?
  createdAt               DateTime            @default(now())
  updatedAt               DateTime            @updatedAt

  @@unique([tenantId, name])
  @@index([tenantId])
  @@index([isActive])
  @@index([isSystem])
  @@index([coaAccountId])
}
